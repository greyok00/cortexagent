# Pathway — Request-Chain Visualization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a capture layer (`lib/pathway.py`) that records every stage a prompt passes through — frame → minify → domain-check → route → generate → beautify — as a per-run NDJSON event stream, plus a hot-memory "load last prompt" source, so the expanded overseer dashboard can render the full request chain.

**Architecture:** A self-contained `lib/pathway.py` module owns an append-only NDJSON store under `~/.cortexagent/pathway/<run_id>.jsonl`. A `PathwayRun` context manager emits one JSON line per stage (6 groups, 12 nodes). The stages this session owns (frame, domain, beautify) are instrumented inside their existing lib functions; minify/route/generate are emitted by the other session and rendering is theirs. Every emit is non-fatal so pathway can never break a request.

**Tech Stack:** Python 3 stdlib (sqlite3 not needed), JSON NDJSON append, reuses `lib/memory_thin.py` (hot memory) and `lib/domain_db.py` (per-domain FTS5+vec0 search).

## Global Constraints

- **Do NOT touch** `lib/overseer.py`, `lib/grammar_proxy.py`, `lib/tray_dashboard.py`, `lib/overseer_dashboard/*`. The other session owns minify/route/generate emission + dashboard rendering. This plan only edits `lib/pathway.py`, `lib/prompt_framing.py`, `lib/beautify.py`, and the spec/plan docs.
- **Never break a request.** All pathway integration is wrapped in try/except; a pathway failure is a silent no-op.
- Localhost-only; no PII in committed files.
- NDJSON idiom matches `lib/memory_thin.py` / `lib/scheduler/store.py` (atomic O_APPEND).
- Domain DBs live in `CFG.state_dir / "domains"` (`~/.cortexagent/domains`); pathway store in `CFG.state_dir / "pathway"`.
- `classify_domain()` may return `professional`/`cybersecurity`, which are NOT in `domain_db.ALLOWED_DOMAINS` (`business, dfir, law, osint, programming`) — `domain_db.search()` already returns `[]` for unknown domains, so check_domain degrades to memory-only gracefully.

```python
#!/usr/bin/env python3
"""lib/pathway.py — request-chain ("pathway") capture layer.

Records every stage a prompt passes through — frame → minify → domain-check →
route → generate → beautify — as one JSON line per stage, appended to
~/.cortexagent/pathway/<run_id>.jsonl. A viewer (the expanded overseer
dashboard) renders these as a clickable node strip.

Emission is non-fatal: a pathway failure is swallowed so it can never break a
request. The NDJSON idiom matches lib/memory_thin.py and lib/scheduler/store.py.

Usage:
  python3 lib/pathway.py --smoke
"""
from __future__ import annotations

import hashlib
import json
import sys
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional

_REPO_ROOT = Path(__file__).resolve().parent.parent
if str(_REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(_REPO_ROOT))

from lib.config import CFG  # noqa: E402

PATHWAY_DIR = CFG.state_dir / "pathway"
MAX_FILES = 200
CURRENT: Optional["PathwayRun"] = None


def _run_id(prompt: str) -> str:
    h = hashlib.sha1(prompt.encode("utf-8", "replace")).hexdigest()[:8]
    return f"{datetime.now():%Y%m%d-%H%M%S}-{h}"


def _rotate() -> None:
    """Trim the pathway dir to MAX_FILES newest files (by mtime)."""
    try:
        if not PATHWAY_DIR.exists():
            return
        files = sorted(PATHWAY_DIR.glob("*.jsonl"),
                       key=lambda p: (p.stat().st_mtime, p.name), reverse=True)
        for stale in files[MAX_FILES:]:
            stale.unlink(missing_ok=True)
    except Exception:
        pass  # rotation is best-effort


class PathwayRun:
    """A single request's pathway. Emits stage events to a per-run JSONL file."""

    def __init__(self, prompt: str):
        self.run_id = _run_id(prompt)
        self.prompt = prompt
        self.seq = 0
        self._path = PATHWAY_DIR / f"{self.run_id}.jsonl"
        _rotate()

    def emit(self, group: str, stage: str, *, status: str = "ok",
             entered: Optional[Dict[str, Any]] = None,
             transformed: Optional[Dict[str, Any]] = None,
             exited: Optional[Dict[str, Any]] = None,
             dur_ms: float = 0.0, tokens_in: int = 0, tokens_out: int = 0) -> None:
        """Append one stage event. Never raises (non-fatal)."""
        try:
            PATHWAY_DIR.mkdir(parents=True, exist_ok=True)
            event = {
                "run_id": self.run_id, "seq": self.seq, "group": group,
                "stage": stage, "status": status,
                "entered": entered or {}, "transformed": transformed or {},
                "exited": exited or {}, "dur_ms": dur_ms,
                "tokens_in": tokens_in, "tokens_out": tokens_out,
                "ts": datetime.now().isoformat(timespec="milliseconds"),
            }
            with open(self._path, "a", encoding="utf-8") as f:
                f.write(json.dumps(event, ensure_ascii=False, default=str) + "\n")
            self.seq += 1
        except Exception:
            pass  # a pathway failure must never break a request

    def __enter__(self) -> "PathwayRun":
        return self

    def __exit__(self, *exc) -> bool:
        return False


def list_runs(n: int = 20) -> List[Dict[str, Any]]:
    """Recent runs, newest first, with first-stage prompt preview."""
    if not PATHWAY_DIR.exists():
        return []
    files = sorted(PATHWAY_DIR.glob("*.jsonl"),
                   key=lambda p: p.stat().st_mtime, reverse=True)[:n]
    out: List[Dict[str, Any]] = []
    for f in files:
        run_id = f.stem
        preview = ""
        ts = ""
        try:
            for line in f.read_text(encoding="utf-8").splitlines():
                try:
                    ev = json.loads(line)
                except Exception:
                    continue
                entered = ev.get("entered") or {}
                if ev.get("seq") == 0:
                    preview = entered.get("prompt") or ""
                ts = ev.get("ts", "")
                break
        except Exception:
            pass
        out.append({"run_id": run_id, "prompt": preview, "ts": ts})
    return out


def read_run(run_id: str) -> Optional[Dict[str, Any]]:
    """Return the run's ordered stages, skipping malformed lines."""
    p = PATHWAY_DIR / f"{run_id}.jsonl"
    if not p.exists():
        return None
    stages: List[Dict[str, Any]] = []
    try:
        for line in p.read_text(encoding="utf-8").splitlines():
            try:
                stages.append(json.loads(line))
            except Exception:
                continue  # tolerate a partial/corrupt line from a crashed run
    except Exception:
        return None
    return {"run_id": run_id, "stages": stages}


def main() -> int:
    if len(sys.argv) > 1 and sys.argv[1] == "--smoke":
        return _smoke()
    print("Usage: python3 lib/pathway.py --smoke")
    return 0


def _smoke() -> int:
    # implemented in Task 6
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def recent_prompts(n: int = 20) -> List[Dict[str, Any]]:
    """User input prompts from hot memory — the "load last prompt" dropdown source.

    Filters to role == "user" with non-empty content. run_id is best-effort:
    matched against recent runs by first-stage prompt when cheap, else None.
    """
    try:
        from lib import memory_thin
        rows = memory_thin.read_all()
    except Exception:
        return []
    users = [r for r in rows
             if r.get("role") == "user" and r.get("content")]
    recent = [{"prompt": r["content"], "ts": r.get("timestamp"), "run_id": None}
              for r in users[-n:]]
    # best-effort run_id correlation against the most recent runs
    try:
        runs = list_runs(n)
        by_prompt = {r["prompt"]: r["run_id"] for r in runs}
        for item in recent:
            item["run_id"] = by_prompt.get(item["prompt"])
    except Exception:
        pass
    return recent
```

```python
def check_domain(domain: str, query: str, limit: int = 10,
                 run: Optional["PathwayRun"] = None) -> List[Dict[str, str]]:
    """Always-run domain-db check: domain DB (FTS5+vec0) + CortexLLM hot memory.

    Returns combined hits [{"tier","source","text"}]. Emits domain.search and
    domain.memory events onto `run` when provided. Any backend failure degrades
    to fewer hits — never raises, never blocks the request.
    """
    hits: List[Dict[str, str]] = []
    db_rows: List[Dict[str, str]] = []
    try:
        from lib.domain_db import search as _db_search
        db_rows = _db_search(domain, query, limit=limit)
        hits = [{"tier": "domain", "source": r.get("source", domain),
                 "text": r.get("chunk", "")} for r in db_rows]
    except Exception:
        db_rows = []
    if run:
        run.emit("domain", "search",
                 status="ok" if db_rows is not None else "skipped",
                 entered={"query": query, "domain": domain},
                 transformed={"hits": len(hits),
                              "top_source": hits[0]["source"] if hits else None},
                 exited={"context_len": sum(len(h["text"]) for h in hits)})
    mem_hits: List[Dict[str, str]] = []
    try:
        from cortexllm.engine import search as _search
        for hit in _search(query, tier="hot", platform="cortexagent", limit=limit):
            content = hit.get("content", "")
            if content:
                mem_hits.append({"tier": "hot", "source": "memory", "text": content})
    except Exception:
        mem_hits = []
    if run:
        run.emit("domain", "memory",
                 entered={"query": query, "tier": "hot"},
                 transformed={"hits": len(mem_hits)},
                 exited={"context_len": sum(len(m["text"]) for m in mem_hits)})
    return hits + mem_hits
```

```python
def _frame_emit(run, stage, **fields):
    """Emit one frame event; non-fatal."""
    try:
        run.emit("frame", stage, **fields)
    except Exception:
        pass


def frame_prompt(prompt, system_prompt, domain=None):
    """Full framing pass: classify, optimize, shrink, and add framing.

    Also starts the pathway run for this request (pathway.CURRENT) and runs the
    always-on domain-db check, so the request chain is captured end to end.
    Returns: (optimized_prompt, framed_system_prompt, domain)
    """
    from lib import pathway  # lazy import — pathway is optional
    run = None
    try:
        run = pathway.PathwayRun(prompt)
        pathway.CURRENT = run
    except Exception:
        run = None

    # 1. Classify domain
    if not domain:
        domain = classify_domain(prompt)
        _frame_emit(run, "classify", entered={"prompt": prompt, "domain": domain},
                    transformed={"domain": domain})

    # 2. Optimize prompt
    optimized = optimize_prompt(prompt)
    _frame_emit(run, "optimize",
                entered={"prompt": prompt},
                transformed={"optimized": optimized,
                             "char_len": len(optimized) - len(prompt)})

    # 3. Shrink if needed
    optimized = shrink_prompt(optimized)

    # 4. Add domain framing to system prompt
    framed_system = add_domain_framing(system_prompt, domain)
    _frame_emit(run, "frame",
                entered={"system_len": len(system_prompt)},
                transformed={"domain": domain},
                exited={"system_len": len(framed_system)})

    # 5. Always-on domain-db check (domain.search + domain.memory events)
    if run is not None:
        try:
            pathway.check_domain(domain, optimized, run=run)
        except Exception:
            pass  # non-fatal

    return optimized, framed_system, domain
```

```python
def _smoke() -> int:
    """Self-test for the pathway capture layer. Runs in an isolated temp dir."""
    import shutil
    import tempfile
    tmp = Path(tempfile.mkdtemp())
    old_dir = PATHWAY_DIR
    globals()["PATHWAY_DIR"] = tmp / "pathway"
    fails = 0

    def _check(name, cond):
        nonlocal fails
        if cond:
            print(f"  ✅ {name}")
        else:
            print(f"  ❌ {name}")
            fails += 1

    # 1. A synthetic run emits exactly 12 stage events across 6 groups.
    groups = ["frame", "minify", "domain", "route", "generate", "beautify"]
    with PathwayRun("synthetic smoke prompt") as run:
        for g in groups:
            run.emit(g, "a")
            run.emit(g, "b")
    stages = read_run(run.run_id)["stages"]
    _check("12 stage events", len(stages) == 12)
    _check("6 groups present",
           {s["group"] for s in stages} == set(groups))

    # 2. Rotation: >200 files trims to MAX_FILES.
    globals()["MAX_FILES"] = 3
    for i in range(5):
        r = PathwayRun(f"rot-{i}")
        r.emit("frame", "classify")
    n_files = len(list((tmp / "pathway").glob("*.jsonl")))
    _check(f"rotation trims to {MAX_FILES} (have {n_files})", n_files <= 3)

    # 3. recent_prompts returns only role==user hot-memory entries.
    from lib import memory_thin
    monkey_tmp = tmp / "hot.jsonl"
    old_hot = memory_thin.HOT_FILE
    memory_thin.HOT_FILE = monkey_tmp
    monkey_tmp.write_text(
        '{"role":"user","content":"p1","timestamp":"1"}\n'
        '{"role":"system","content":"sys","timestamp":"2"}\n'
        '{"role":"user","content":"p2","timestamp":"3"}\n')
    prompts = recent_prompts(n=10)
    _check("recent_prompts filters to user prompts",
           [p["prompt"] for p in prompts] == ["p1", "p2"])
    memory_thin.HOT_FILE = old_hot

    # 4. Malformed JSONL lines are tolerated on read.
    bad = PathwayRun("bad")
    bad.emit("frame", "classify")
    with open(PATHWAY_DIR / f"{bad.run_id}.jsonl", "a") as f:
        f.write("{nope}\n")
    _check("read_run tolerates malformed lines",
           len(read_run(bad.run_id)["stages"]) == 1)

    # 5. emit failure is non-fatal (a raised/unserializable stage does not propagate).
    sf = PathwayRun("fail")
    sf.emit("frame", "classify", entered={"bad": object()})
    sf.emit("frame", "optimize")
    _check("emit failure is non-fatal",
           len(read_run(sf.run_id)["stages"]) == 1)

    # 6. check_domain degrades gracefully (unknown domain -> memory-only, emits both).
    cd = PathwayRun("cd")
    hits = check_domain("professional", "review this", run=cd)
    cd_groups = [s["stage"] for s in read_run(cd.run_id)["stages"]]
    _check("check_domain emits search+memory",
           cd_groups == ["search", "memory"] and isinstance(hits, list))

    shutil.rmtree(tmp, ignore_errors=True)
    globals()["PATHWAY_DIR"] = old_dir
    print("pathway smoke PASS" if fails == 0 else f"❌ {fails} failures")
    return 1 if fails else 0
```
